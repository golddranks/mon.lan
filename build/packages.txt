# Extra packages baked into the image.
#
# Target/profile default packages are included automatically; prefix an entry
# with '-' to drop a default. Audit the result in the generated *.manifest.

# --- Web UI (not part of the default package set; a bare image is CLI-only) ---
luci
luci-ssl

# --- WiFi: full hostapd (802.11r/AP/WPS) instead of the basic default ---
-wpad-basic-mbedtls
wpad-mbedtls
hostapd-utils

# --- Diagnostics / convenience ---
curl
nano
bind-dig
tcpdump
ip-full
iperf3

# --- IPv6-only LAN: NAT64 (Jool) + DNS64 (unbound) ---
# The kmod package autoloads the Jool kernel modules; jool-tools ships the
# control tool, init, and /etc/config/jool. unbound is the DNS64 resolver
# behind dnsmasq (dnsmasq itself cannot synthesise AAAA records).
kmod-jool-netfilter
jool-tools-netfilter
unbound-daemon

# --- Dynamic DNS (Gandi LiveDNS; config in /etc/config/ddns) ---
ddns-scripts
ddns-scripts-gandi

# --- TLS front-end: nginx serves LuCI via uwsgi (see /etc/config/nginx) ---
# nginx-mod-luci's uci-defaults disables uhttpd on first boot.
nginx-ssl
nginx-mod-luci

# --- Let's Encrypt, DNS-01 via the Gandi API (see /etc/config/acme) ---
acme-common
acme-acmesh
acme-acmesh-dnsapi
