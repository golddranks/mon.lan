# Extra packages baked into the Archer C7 image.
#
# Target/profile default packages are included automatically.
# Prefix an entry with '-' to drop a default.
# Audit the result in the generated *.manifest.

# --- Web UI (not part of the default package set; a bare image is CLI-only) ---
luci
luci-ssl

# --- mon is the only DHCP and DNS server on this LAN ---
# A second dnsmasq would answer nothing useful, and a second odhcpd would send
# its own RAs with no prefix to advertise.
-dnsmasq
-odhcpd-ipv6only

# --- WiFi: full hostapd, same as mon, so 802.11k/r/v are actually available ---
# The default wpad-basic has no k/v support, and enabling them there disables
# the interface.
-wpad-basic-mbedtls
wpad-mbedtls
# Steering and neighbour-report sync between the APs (see /etc/config/usteer).
usteer

# --- Unprivileged probes ----------------------------------------------------
# busybox ping is raw-socket only and refuses non-root; iputils' ping uses ICMP
# datagram sockets, which the ping group range in
# /etc/sysctl.d/20-ping-group.conf allows for ghost.
iputils-ping
